func readSample(c *sql.Conn,q string,deadline time.Duration)(float64,int,string,string){
 ctx,cancel:=context.WithTimeout(context.Background(),deadline);defer cancel();t:=time.Now();rows,e:=c.QueryContext(ctx,q);if e!=nil{return float64(time.Since(t).Nanoseconds())/1e6,0,"",e.Error()};defer rows.Close();cols,e:=rows.Columns();must(e);n:=0;h:=sha256.New()
 for rows.Next(){v:=make([]any,len(cols));p:=make([]any,len(cols));for j:=range v{p[j]=&v[j]};if e=rows.Scan(p...);e!=nil{break};fmt.Fprintln(h,v);n++};if e==nil{e=rows.Err()};err:="";if e!=nil{err=e.Error()};return float64(time.Since(t).Nanoseconds())/1e6,n,hex.EncodeToString(h.Sum(nil)),err
}
func readCohort(c *sql.Conn,q string,samples int,budget time.Duration)map[string]any{
 good:=[]float64{};bad:=[]float64{};errors:=map[string]int{};var count int;var hash string
 for j:=0;j<samples;j++{ms,n,h,e:=readSample(c,q,budget);if e!=""{bad=append(bad,ms);errors[e]++}else{good=append(good,ms);count=n;hash=h}}
 result:=map[string]any{"samples_ms":good,"refused_ms":bad,"errors":errors,"completed":len(good),"refused":len(bad),"rows":count,"row_hash":hash,"sql":q,"budget_ms":budget.Milliseconds()}
 if len(good)>0{result["p50_ms"]=pct(good,.5);result["p95_ms"]=pct(good,.95);result["p99_ms"]=pct(good,.99)};if len(bad)>0{result["refusal_p95_ms"]=pct(bad,.95);result["refusal_max_ms"]=pct(bad,1)};return result
}

func bandBrowse(field string,ceiling int,sortcol,exclude string)string{
 parts:=[]string{};for _,band:=range []int{-1,0,8,13,17,18}{if band>ceiling{continue};where:=fmt.Sprintf("e.library_id=1 AND e.kind=1 AND e.%s=%d",field,band);if exclude!=""{where+=" AND "+exclude};parts=append(parts,fmt.Sprintf("SELECT * FROM(SELECT e.id,e.%s AS value FROM catalog_entities e WHERE %s ORDER BY e.%s DESC,e.id DESC LIMIT40)",sortcol,where,sortcol))};q:="SELECT * FROM("+strings.Join(parts," UNION ALL ")+") ORDER BY value DESC,id DESC LIMIT40";return strings.ReplaceAll(q,"LIMIT40","LIMIT 40")
}
func bandDeltaSQL(item,old,next int)string{
 q:="";for _,mode:=range []int{0,1}{q+=fmt.Sprintf("UPDATE rating_band_facet_counts SET total=total-1 WHERE library_id=1 AND kind=1 AND mode=%d AND rating_band=%d AND term_id IN(SELECT term_id FROM catalog_entity_terms WHERE entity_id=%d);",mode,old,item)
 q+=fmt.Sprintf("INSERT INTO rating_band_facet_counts SELECT1,1,term_id,%d,%d,1 FROM catalog_entity_terms WHERE entity_id=%d ON CONFLICT(library_id,kind,term_id,mode,rating_band) DO UPDATE SET total=total+1;",mode,next,item)};return strings.ReplaceAll(q,"SELECT1","SELECT 1")
}
