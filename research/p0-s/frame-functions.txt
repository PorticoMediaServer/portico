func sourceRows(streams,chapters,book int,tag string)string{
 var q strings.Builder
 for j:=0;j<streams;j++{typ,codec,channels:="audio","aac",2;if j>=255&&j<510{typ="video";codec="h264";channels=0};if j>=510{typ="subtitle";codec="subrip";channels=0};fmt.Fprintf(&q,"INSERT INTO asset_streams(asset_id,stream_index,type,codec,language,title,channels,channel_layout,is_default,is_forced,has_asset_stream_breadth) VALUES('asset-1',%d,'%s','%s','en','%s-%d',%d,'stereo',0,0,1);",j,typ,codec,tag,j,channels)}
 for j:=0;j<chapters;j++{fmt.Fprintf(&q,"INSERT INTO asset_chapters VALUES('asset-1',%d,'%s-%d',%d,%d);",j,tag,j,j*60,(j+1)*60)}
 for j:=0;j<book;j++{fmt.Fprintf(&q,"INSERT INTO catalog_book_chapters(file_id,chapter_index,title,start_seconds,end_seconds) VALUES(1,%d,'%s-%d',%d,%d);",j,tag,j,j*60,(j+1)*60)}
 return q.String()
}
func versionRows(n,offset int)string{
 var q strings.Builder;for j:=0;j<n;j++{fmt.Fprintf(&q,"INSERT INTO catalog_asset_links(entity_id,asset_id,library_id) VALUES(1,%d,1);",10001+offset+j)};return q.String()
}
